// Conversational State Machine with Voice Support let chatState = { step: 'GREETING', userName: '', userLocation: '', userIntent: '' }; // Toggle Web Speech Recognition (Speech-to-Text) function toggleVoiceInput() { const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition; if (!SpeechRecognition) { alert("Speech recognition is not supported in this browser. Please use Chrome or Edge."); return; } const recognition = new SpeechRecognition(); const langSelect = document.getElementById('chatLanguage'); recognition.lang = langSelect ? langSelect.value : 'en-US'; const micBtn = document.getElementById('micBtn'); micBtn.classList.add('text-red-400', 'animate-pulse'); recognition.onresult = function(event) { const speechToText = event.results[0][0].transcript; document.getElementById('chatInput').value = speechToText; micBtn.classList.remove('text-red-400', 'animate-pulse'); }; recognition.onerror = function() { micBtn.classList.remove('text-red-400', 'animate-pulse'); }; recognition.onend = function() { micBtn.classList.remove('text-red-400', 'animate-pulse'); }; recognition.start(); } // Text-to-Speech (AI Voice Output) function speakResponse(text) { if ('speechSynthesis' in window) { const cleanText = text.replace(/<[^>]*>?/gm, ''); const utterance = new SpeechSynthesisUtterance(cleanText); const langSelect = document.getElementById('chatLanguage'); utterance.lang = langSelect ? langSelect.value : 'en-US'; window.speechSynthesis.speak(utterance); } } async function handleChatQuery() { const chatInput = document.getElementById('chatInput'); const chatBox = document.getElementById('chatBox'); const query = chatInput.value.trim(); if (!query) return; chatBox.innerHTML += `
${escapeHtml(query)}
`; chatInput.value = ''; chatBox.scrollTop = chatBox.scrollHeight; setTimeout(async () => { const reply = await generateAdvancedAgentResponse(query); chatBox.innerHTML += `
${reply}
`; chatBox.scrollTop = chatBox.scrollHeight; speakResponse(reply); }, 400); } async function generateAdvancedAgentResponse(query) { const upperQuery = query.toUpperCase(); if (chatState.step === 'ASK_NAME') { chatState.userName = query.trim(); chatState.step = 'ASK_LOCATION'; return `Nice to meet you, ${escapeHtml(chatState.userName)}! May I know your current location or city?`; } if (chatState.step === 'ASK_LOCATION') { chatState.userLocation = query.trim(); chatState.step = 'ASK_INTENT'; return `Got it, from ${escapeHtml(chatState.userLocation)}. What is the main purpose today? (e.g., checking shipment SHIP001, pincode, or route)`; } if (chatState.step === 'ASK_INTENT') { chatState.userIntent = query.trim(); chatState.step = 'READY'; return `Thank you, ${escapeHtml(chatState.userName)}. I'm ready! You can now ask about tracking IDs like SHIP001 or delivery pincodes.`; } if (upperQuery.includes('SHIP')) { const shipments = await DB.fetchShipments(); const found = shipments.find(s => upperQuery.includes(s.tracking_id.toUpperCase())); if (found) { return `📦 Shipment (${found.tracking_id}): Status: ${found.status}, Carrier: ${found.carrier_id}, Location: ${found.current_location}`; } else { return `❌ Sorry ${escapeHtml(chatState.userName)}, tracking ID not found. Try asking for SHIP001.`; } } if (/\d{6}/.test(query)) { const pinMatch = query.match(/\d{6}/)[0]; const carriers = await DB.fetchCarriers(); const servicing = carriers.filter(c => c.serviced_pincodes.includes(pinMatch)); if (servicing.length > 0) { const names = servicing.map(c => c.carrier_name).join(', '); return `✅ Pincode ${pinMatch} is serviced by: ${names}.`; } else { return `⚠️ Pincode ${pinMatch} has limited coverage.`; } } return `How else can I help you today, ${escapeHtml(chatState.userName)}?`; } document.addEventListener('DOMContentLoaded', () => { const chatBox = document.getElementById('chatBox'); if (chatBox) { const hour = new Date().getHours(); let timeGreeting = hour < 12 ? "Good Morning" : (hour < 18 ? "Good Afternoon" : "Good Evening"); setTimeout(() => { const greetingText = `🤖 ${timeGreeting}! Welcome to AxomOps AI Logistics. May I know your name please?`; chatBox.innerHTML += `
${greetingText}
`; chatState.step = 'ASK_NAME'; speakResponse(greetingText); }, 500); } }); function escapeHtml(text) { const map = { '&': '&', '<': '<', '>': '>', '"': '"', "'": ''' }; return text.replace(/[&<>"']/g, function(m) { return map[m]; }); }
